# properly initializes a warp
# called from warp_drive.100 or warp_drive.101, executes when a jump flag is set and both source and destination targets exist
fleet_event = {
	id = warp_plugin.102
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		fleet_has_warp_drive = yes
		has_fleet_flag = warp_plugin_drive_active
		exists = event_target:warp_plugin_source
		exists = event_target:warp_plugin_destination
	}

	immediate = {
		# get distance from jump source to jump destination
		export_trigger_value_to_variable = {
			trigger = distance
			variable = warp_distance_total
			parameters = {
				source = event_target:warp_plugin_source
				type = euclidean
			}
		}
		warp_plugin_get_warp_speed = {
			FLEET = this
			OUTPUT = warp_speed_mult
		}
		set_variable = {
			which = warp_time_remaining
			value = value:warp_travel_time|DISTANCE|warp_distance_total|SPEED|warp_speed_mult|
		}
		set_variable = {
			which = warp_time_total
			value = warp_time_remaining
		}
		clear_variable = warp_distance_total
		clear_variable = warp_speed_mult
		# start situation
		owner = {
			start_situation = {
				type = situation_warp_in_progress
				target = root
			}
		}
		warp_plugin_retrieve_warp_situation = {
			FLEET = root
			TARGET = warp_situation
		}
		# final setup, set relevant flags and modifiers
		set_event_locked = yes
		add_modifier = {
			modifier = warp_plugin_warp_in_progress
		}
		set_saved_date = {
			key = warp_plugin_arrival_date
			days_from_present = warp_time_remaining
			expires = warp_time_remaining
		}
		fleet_event = {
			id = warp_plugin.110
			days = 1
		}
	}
}

# pulse event for when ship is warping
# called from warp_plugin.102, then recursively until situation ends or timer reaches 0
fleet_event = {
	id = warp_plugin.110
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = event_target:warp_situation
	}

	immediate = {
		# tick down
		subtract_variable = {
			which = warp_time_remaining
			value = 1
		}

		if = { # warp complete
			limit = {
				check_variable = {
					which = warp_time_remaining
					value = 0
				}
			}
			# randomize fleet location in system and notify owner of arrival
			create_message = {
				type = MESSAGE_TYPE_WARP_ARRIVED
				localization = MESSAGE_WARP_ARRIVED
				days = 30
				target = this
				variable = {
					type = name
					localization = FLEET_NAME
					scope = this
				}
				variable = {
					type = name
					localization = SYSTEM
					scope = solar_system
				}
			}
			set_location = {
				target = event_target:warp_plugin_destination
				distance = @warp_entry_distance
				angle = random
				direction = in_system
			}
			# common cleanup
			warp_plugin_warp_end_cleanup = {
				FLEET = root
				TARGET = event_target:warp_situation
			}
		}
		else = { # update situation
			event_target:warp_situation = {
				add_situation_progress = progress_rate
			}
			fleet_event = {
				id = warp_plugin.110
				days = 1
			}
		}
	}
}

# called when fleet enters mia
fleet_event = {
	id = warp_plugin.111
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_fleet_order = experimental_subspace_navigation_fleet_order } # ESN should still process normally
		OR = { # Nomads do not own systems and will have to use the locations of their capital arks instead
			AND = {
				exists = root.starbase
				exists = root.starbase.solar_system
			}
			AND = {
				exists = root.owner.capital_scope
				exists = root.owner.capital_scope.solar_system
			}
		}
	}

	immediate = {
		# get home base, using capital as fallback
		if = {
			limit = {
				exists = root.starbase
				exists = root.starbase.solar_system
			}
			root.starbase.fleet.solar_system = {
				save_event_target_as = warp_plugin_home_base
			}
		}
		else = {
			root.owner.capital_scope.solar_system = {
				save_event_target_as = warp_plugin_home_base
			}
		}
		# find previous location and compute distance
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = root
			FTYPE = fleet
			FLAG = warp_plugin_location
			OUTPUT = warp_plugin_location
		}
		# get the shortest path (using either end of a path, provided the previous location is set)
		event_target:warp_plugin_location = {
			export_trigger_value_to_variable = {
				trigger = distance
				variable = warp_distance_total
				parameters = {
					source = event_target:warp_plugin_home_base
					type = euclidean
				}
			}
			root = {
				set_variable = {
					which = warp_distance_total
					value = prev.warp_distance_total
				}
			}
			clear_variable = warp_distance_total
		}
		# compute time needed to reach destination
		warp_plugin_get_warp_speed = {
			FLEET = this
			OUTPUT = warp_speed_mult
		}
		# add to warp time if one exists, otherwise set warp time
		if = {
			limit = {
				is_variable_set = warp_time_remaining
			}
			change_variable = {
				which = warp_time_remaining
				value = value:warp_travel_time|DISTANCE|warp_distance_total|SPEED|warp_speed_mult|
			}
		}
		else = {
			set_variable = {
				which = warp_time_remaining
				value = value:warp_travel_time|DISTANCE|warp_distance_total|SPEED|warp_speed_mult|
			}
		}
		clear_variable = warp_distance_total
		clear_variable = warp_speed_mult
		# make mia
		warp_plugin_initiate_return_mia = {
			SCOPE = this
			VALUE = warp_time_remaining
		}
		set_saved_date = {
			key = warp_plugin_arrival_date
			days_from_present = warp_time_remaining
			expires = warp_time_remaining
		}
		create_message = {
			type = MESSAGE_TYPE_WARP_ABORTED
			localization = MESSAGE_WARP_ABORTED
			days = 30
			target = this
			variable = {
				type = name
				localization = FLEET_NAME
				scope = this
			}
			variable = {
				type = name
				localization = SYSTEM
				scope = event_target:warp_plugin_location
			}
			variable = {
				type = variable
				varname = warp_time_remaining
				localization = DAYS
				scope = this
			}
		}
		# common cleanup
		warp_plugin_retrieve_warp_situation = {
			FLEET = root
			TARGET = warp_situation
		}
		warp_plugin_warp_end_cleanup = {
			FLEET = root
			TARGET = event_target:warp_situation
		}
	}
}
