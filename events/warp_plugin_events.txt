
# warp_plugin - general events
# id: 0-10000
namespace = warp_plugin

# called on start and single player save load
# if added after multiplayer start, blame the host for indecision
event = {
	id = warp_plugin.0
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		every_country = {
			# if leader is a proponent, country becomes adherents
			limit = {
				exists = leader
				leader = {
					has_trait = leader_trait_warp_proponent
				}
			}
			force_add_civic = civic_warp_ftl_adherents
			refresh_auto_generated_ship_designs = yes
		}

		set_global_flag = warp_plugin_installed
	}
}

# called when a new country is created (on_country_created, on_release_as_vassal, and on_country_released_in_war
country_event = {
	id = warp_plugin.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		country_prefers_warp_drives = no
	}
	
	immediate = {
		# All three on-actions currently have the same scopes:
		# THIS = created country
		# FROM = country scope where THIS is created (creator, overlord, or 'liberator')
		if = {
			limit = {
				OR = {
					# if leader is a proponent, country becomes adherents
					AND = {
						exists = leader
						leader = {
							has_trait = leader_trait_warp_proponent
						}
					}
					# if creator is adherent, country becomes adherents
					AND = {
						exists = from
						from = {
							country_prefers_warp_drives = yes
						}
					}
				}
			}
			force_add_civic = civic_warp_ftl_adherents
			refresh_auto_generated_ship_designs = yes
			country_event = {
				id = warp_plugin.1
			}
		}
	}
}

## jump events
# triggers when the fleet jumps (on_jump_drive)
ship_event = {
	id = warp_plugin.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		# prevent multiple ships within the same fleet from calling this method
		# only 1 ship is needed to trigger the drive
		fleet = {
			NOT = {
				has_fleet_flag = warp_plugin_drive_active
			}
		}
	}

	immediate = {
		fleet = {
			# set drive flag for the fleet
			set_timed_fleet_flag = {
				flag = warp_plugin_drive_active
				days = 1
			}
			# attempt to call warp_plugin.103
			fleet_event = {
				id = warp_plugin.103
			}
		}
	}
}

# triggers when fleet arrives in a system (on_entering_system_fleet)
fleet_event = {
	id = warp_plugin.101
	hide_window = yes
	is_triggered_only = yes

	# Scope = Fleet
	# From = System
	immediate = {
		# get current location and clear its flag as it is outdated
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = this
			FTYPE = fleet
			FLAG = warp_plugin_current_location
			OUTPUT = warp_plugin_outdated_location
		}
		if = {
			limit = {
				exists = event_target:warp_plugin_outdated_location
			}
			remove_fleet_flag = warp_plugin_current_location_@event_target:warp_plugin_outdated_location
		}
		# mark current system asw current location
		set_fleet_flag = warp_plugin_current_location_@from
		# set location change flag
		set_timed_fleet_flag = {
			flag = warp_plugin_current_location_changed
			days = 1
		}
		# attempt to call warp_plugin.103
		fleet_event = {
			id = warp_plugin.103
		}
	}
}

# triggers when fleet leaves a system (on_leaving_system_fleet)
fleet_event = {
	id = warp_plugin.102
	hide_window = yes
	is_triggered_only = yes

	# Scope = Fleet
	# From = System
	immediate = {
		# get prev-prev location and clear its flag as it is now outdated
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = this
			FTYPE = fleet
			FLAG = warp_plugin_prev_prev_location
			OUTPUT = warp_plugin_outdated_location
		}
		if = {
			limit = {
				exists = event_target:warp_plugin_outdated_location
			}
			remove_fleet_flag = warp_plugin_prev_prev_location_@event_target:warp_plugin_outdated_location
		}
		# get prev location and re-mark it as prev-prev
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = this
			FTYPE = fleet
			FLAG = warp_plugin_prev_location
			OUTPUT = warp_plugin_prev_prev_location
		}
		if = {
			limit = {
				exists = event_target:warp_plugin_prev_prev_location
			}
			set_fleet_flag = warp_plugin_prev_prev_location_@event_target:warp_plugin_prev_prev_location
			remove_fleet_flag = warp_plugin_prev_location_@event_target:warp_plugin_prev_prev_location
		}
		# mark last system as prev
		set_fleet_flag = warp_plugin_prev_location_@from
		# set location change flag
		set_timed_fleet_flag = {
			flag = warp_plugin_prev_location_changed
			days = 1
		}
		# attempt to call warp_plugin.103
		fleet_event = {
			id = warp_plugin.103
		}
		# attempt to call warp_plugin.102, in case the fleet left due to an mia
		fleet_event = {
			id = warp_plugin.121
		}
	}
}

# properly initializes a warp
fleet_event = {
	id = warp_plugin.103
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		fleet_has_warp_drive = yes # at least one ship in the jumping fleet must have a warp drive (it is already required that all ships can jump)
		has_fleet_flag = warp_plugin_drive_active # warp_plugin.100
		has_fleet_flag = warp_plugin_current_location_changed # warp_plugin.101
		has_fleet_flag = warp_plugin_prev_location_changed # warp_plugin.102
	}

	immediate = {
		# obtain the required scopes as targets
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = this
			FTYPE = fleet
			FLAG = warp_plugin_prev_location
			OUTPUT = warp_plugin_source
		}
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = this
			FTYPE = fleet
			FLAG = warp_plugin_current_location
			OUTPUT = warp_plugin_destination
		}
		# get distance from jump source to jump destination
		export_trigger_value_to_variable = {
			trigger = distance
			variable = warp_plugin_distance_total
			parameters = {
				source = event_target:warp_plugin_source
				type = euclidean
			}
		}
		warp_plugin_get_warp_speed = {
			FLEET = this
			OUTPUT = warp_plugin_speed_mult
		}
		set_variable = {
			which = warp_plugin_time_remaining
			value = value:warp_travel_time|DISTANCE|warp_plugin_distance_total|SPEED|warp_plugin_speed_mult|
		}
		set_variable = {
			which = warp_plugin_time_total
			value = warp_plugin_time_remaining
		}
		log = "[This.GetName] travel distance: [This.warp_plugin_distance_total], speed: [This.warp_plugin_speed_mult], time: [This.warp_plugin_time_total]"
		# start situation
		owner = {
			start_situation = {
				type = situation_warp_in_progress
				target = root
			}
		}
		warp_plugin_retrieve_warp_situation = {
			FLEET = root
			TARGET = warp_plugin_situation
		}
		# final setup, set relevant flags and modifiers
		set_event_locked = yes
		add_modifier = {
			modifier = warp_plugin_warp_in_progress
		}
		set_saved_date = {
			key = warp_plugin_arrival_date
			days_from_present = warp_plugin_time_remaining
			expires = warp_plugin_time_remaining
		}
		fleet_event = {
			id = warp_plugin.110
			days = 1
		}
	}
}


# pulse event for when ship is warping
# called from warp_plugin.102, then recursively until situation ends or timer reaches 0
fleet_event = {
	id = warp_plugin.110
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = event_target:warp_plugin_situation
	}

	immediate = {
		# tick down
		subtract_variable = {
			which = warp_plugin_time_remaining
			value = 1
		}

		if = { # warp complete
			limit = {
				check_variable = {
					which = warp_plugin_time_remaining
					value = 0
				}
			}
			# randomize fleet location in system and notify owner of arrival
			create_message = {
				type = MESSAGE_TYPE_WARP_ARRIVED
				localization = MESSAGE_WARP_ARRIVED
				days = 30
				target = this
				variable = {
					type = name
					localization = FLEET_NAME
					scope = this
				}
				variable = {
					type = name
					localization = SYSTEM
					scope = solar_system
				}
			}
			set_location = {
				target = event_target:warp_plugin_destination
				distance = @warp_entry_distance
				angle = random
				direction = in_system
			}
			# common cleanup
			warp_plugin_warp_end_cleanup = {
				FLEET = root
				TARGET = event_target:warp_plugin_situation
			}
		}
		else = { # update situation and recursively invoke in a day
			event_target:warp_plugin_situation = {
				add_situation_progress = progress_rate
			}
			fleet_event = {
				id = warp_plugin.110
				days = 1
			}
		}
	}
}

# properly initiates an MIA return
fleet_event = {
	id = warp_plugin.121
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		fleet_has_warp_drive = yes
		has_fleet_flag = warp_plugin_mia_initiated # warp_plugin.120
		has_fleet_flag = warp_plugin_prev_location_changed # warp_plugin.102
		# current location not needed because it currently does not exists; it becomes available upon return from mia
		# make sure fleet can return to some home base or capital; otherwise let the game sort it out
		# note that nomads do not own systems and will have to use the locations of their capital arks instead
	}

	immediate = {
		# get home base, using capital as fallback
		if = {
			limit = {
				exists = root.starbase
				exists = root.starbase.solar_system
			}
			root.starbase.fleet.solar_system = {
				save_event_target_as = warp_plugin_home_base
			}
		}
		else = {
			root.owner.capital_scope.solar_system = {
				save_event_target_as = warp_plugin_home_base
			}
		}
		# find current and previous locations and compute distance
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = root
			FTYPE = fleet
			FLAG = warp_plugin_prev_location
			OUTPUT = warp_plugin_destination
		}
		d225_lib_get_scope_by_flag = {
			TYPE = system
			SCOPE = root
			FTYPE = fleet
			FLAG = warp_plugin_prev_prev_location
			OUTPUT = warp_plugin_source
		}
		# check if warp situation exists; get return distance if it does, otherwise just use prev for distance calc
		warp_plugin_retrieve_warp_situation = {
			FLEET = root
			TARGET = warp_plugin_situation
		}
		if = {
			limit = {
				exists = event_target:warp_plugin_situation
			}
			# compute distance traveled
			set_variable = {
				which = warp_plugin_distance_traveled
				value = value:d225_lib_compute|VAL1|warp_plugin_time_total|OP|subtract|VAL2|warp_plugin_time_remaining|
			}
			multiply_variable = {
				which = warp_plugin_distance_traveled
				value = warp_plugin_distance_total
			}
			divide_variable = {
				which = warp_plugin_distance_traveled
				value = warp_plugin_time_total
			}
			# compute return distance
			warp_plugin_compute_distance_to_home = {
				SRC = event_target:warp_plugin_source
				DEST = event_target:warp_plugin_destination
				HOME = event_target:warp_plugin_home_base
				DIST = warp_plugin_distance_traveled
				OUTPUT = warp_plugin_return_distance
			}
			log = "[This.GetName] distance traveled: [This.warp_plugin_distance_traveled] out of [This.warp_plugin_distance_total]"
			clear_variable = warp_plugin_distance_traveled
		}
		else = {
			# get the return distance from the last location
			event_target:warp_plugin_destination = {
				export_trigger_value_to_variable = {
					trigger = distance
					variable = warp_plugin_return_distance
					parameters = {
						source = event_target:warp_plugin_home_base
						type = euclidean
					}
				}
				root = {
					set_variable = {
						which = warp_plugin_return_distance
						value = prev.warp_plugin_return_distance
					}
				}
				clear_variable = warp_plugin_return_distance
			}
		}
		# compute time needed to reach destination
		warp_plugin_get_warp_speed = {
			FLEET = this
			OUTPUT = warp_plugin_return_speed_mult
		}
		# add to warp time if one exists, otherwise set warp time
		set_variable = {
			which = warp_plugin_return_time_remaining
			value = value:warp_travel_time|DISTANCE|warp_plugin_return_distance|SPEED|warp_plugin_return_speed_mult|
		}
		# make mia
		warp_plugin_initiate_return_mia = {
			SCOPE = this
			VALUE = warp_plugin_return_time_remaining
		}
		set_saved_date = {
			key = warp_plugin_arrival_date
			days_from_present = warp_plugin_return_time_remaining
			expires = warp_plugin_return_time_remaining
		}
		create_message = {
			type = MESSAGE_TYPE_WARP_ABORTED
			localization = MESSAGE_WARP_ABORTED
			days = 30
			target = this
			variable = {
				type = name
				localization = FLEET_NAME
				scope = this
			}
			variable = {
				type = name
				localization = SYSTEM
				scope = event_target:warp_plugin_destination
			}
			variable = {
				type = variable
				varname = warp_plugin_return_time_remaining
				localization = DAYS
				scope = this
			}
		}
		# common cleanup
		warp_plugin_warp_end_cleanup = {
			FLEET = root
			TARGET = event_target:warp_plugin_situation
		}
		log = "[This.GetName] return distance: [This.warp_plugin_return_distance], speed: [This.warp_plugin_return_speed_mult], time: [This.warp_plugin_return_time_remaining]"
		clear_variable = warp_plugin_return_distance
		clear_variable = warp_plugin_return_speed_mult
		clear_variable = warp_plugin_return_time_remaining
	}
}
