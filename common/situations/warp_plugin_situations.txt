
situation_warp_in_progress = {
	picture = GFX_evt_ship_travel
	category = positive

	complete_icon_frame = GFX_situation_outcome_frame_green
	complete_icon = GFX_situation_outcome_positive

	on_start = {
		# monthly rate = 100 / total time * 30
		set_variable = {
			which = total_days
			value = target.warp_plugin_time_total
		}
		set_variable = {
			which = progress_rate
			value = 100
		}
		divide_variable = {
			which = progress_rate
			value = total_days
		}
	}
	on_progress_complete = {
		custom_tooltip = warp_plugin_arrived
		hidden_effect = {
			# abort_situation = this
			set_situation_locked = yes
		}
	}
	on_abort = {
		target = {
			clear_variable = total_days
			clear_variable = progress_rate
		}
		destroy_situation = this
	}
	abort_trigger = {
		OR = {
			NOT = {
				exists = target
			}
			target = {
				not = { is_variable_set = warp_plugin_time_remaining }
			}
		}
	}
	
	progress_direction = monodirectional

	stages = {
		warp_accelerating = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_green
			end = 25
		}
		warp_cruising = {
			icon = GFX_situation_stage_2
			icon_background = GFX_situation_stage_frame_green
			end = 65
		}
		warp_decelerating = {
			icon = GFX_situation_stage_3
			icon_background = GFX_situation_stage_frame_green
			end = 90
		}
		warp_final_approach = {
			icon = GFX_situation_stage_4
			icon_background = GFX_situation_stage_frame_green
			end = 100
		}
	}

	monthly_progress = {
		base = 0
	}

	approach = {
		name = approach_normal_operations
		icon = GFX_situation_approach_fleet
		icon_background = GFX_situation_approach_bg_green
		default = yes
		on_select = {
			custom_tooltip = approach_normal_operations_on_select
		}
	}

	on_monthly = {
	}
}
